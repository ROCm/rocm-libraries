cmake_minimum_required(VERSION 3.25.2)
project(cmake-find-test-clang LANGUAGES CXX)

find_package(Clang QUIET)
if(Clang_FOUND)
    message(STATUS "Clang_FOUND: YES")
    message(STATUS "Clang_PACKAGE_VERSION: '${Clang_PACKAGE_VERSION}'")
    message(STATUS "CLANG_VERSION: '${CLANG_VERSION}'")
    message(STATUS "CLANG_VERSION_MAJOR: '${CLANG_VERSION_MAJOR}'")
    message(STATUS "CLANG_INSTALL_PREFIX: '${CLANG_INSTALL_PREFIX}'")
    message(STATUS "CLANG_TOOLS_BINARY_DIR: '${CLANG_TOOLS_BINARY_DIR}'")
    message(STATUS "CLANG_CMAKE_DIR: '${CLANG_CMAKE_DIR}'")
    message(STATUS "Clang_DIR: '${Clang_DIR}'")

    # Dump all CLANG/Clang variables
    get_cmake_property(_vars VARIABLES)
    message(STATUS "")
    message(STATUS "── All CLANG* variables ──")
    foreach(_var ${_vars})
        if(_var MATCHES "^CLANG|^Clang")
            message(STATUS "  ${_var} = ${${_var}}")
        endif()
    endforeach()

    # Check if any tool paths are discoverable
    message(STATUS "")
    message(STATUS "── Tool search using Clang package info ──")
    foreach(TOOL clang-tidy clang-format run-clang-tidy)
        if(CLANG_INSTALL_PREFIX)
            if(EXISTS "${CLANG_INSTALL_PREFIX}/bin/${TOOL}")
                message(STATUS "  ${TOOL}: EXISTS at ${CLANG_INSTALL_PREFIX}/bin/${TOOL}")
            else()
                message(STATUS "  ${TOOL}: NOT at ${CLANG_INSTALL_PREFIX}/bin/${TOOL}")
            endif()
        else()
            message(STATUS "  ${TOOL}: no CLANG_INSTALL_PREFIX to check")
        endif()
    endforeach()
else()
    message(STATUS "Clang_FOUND: NO")
endif()
