cmake_minimum_required(VERSION 3.20)
project(rocke_native_sdpa LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(ROCKE_BUILD_SHARED_ENGINE ON CACHE BOOL "" FORCE)
set(ROCKE_BUILD_PYBIND OFF CACHE BOOL "" FORCE)
if(NOT ROCKE_PLATFORM)
  message(FATAL_ERROR "Pass -DROCKE_PLATFORM=<pinned source>/rocke/platform")
endif()
add_subdirectory("${ROCKE_PLATFORM}" rocke)
find_path(COMGR_INCLUDE amd_comgr.h HINTS "$ENV{ROCM_PATH}/include" PATH_SUFFIXES amd_comgr REQUIRED)
find_path(HIP_INCLUDE hip/hip_runtime_api.h HINTS "$ENV{ROCM_PATH}/include" REQUIRED)
find_library(COMGR_LIBRARY amd_comgr HINTS "$ENV{ROCM_PATH}/lib" REQUIRED)
find_library(HIP_LIBRARY amdhip64 HINTS "$ENV{ROCM_PATH}/lib" REQUIRED)
find_package(Threads REQUIRED)
add_executable(test_native_sdpa test_native_sdpa.cpp)
target_compile_definitions(test_native_sdpa PRIVATE __HIP_PLATFORM_AMD__)
target_compile_options(test_native_sdpa PRIVATE -Wall -Wextra -Werror)
target_include_directories(test_native_sdpa PRIVATE "${ROCKE_PLATFORM}/cpp/include" "${COMGR_INCLUDE}" "${HIP_INCLUDE}")
target_link_libraries(test_native_sdpa PRIVATE rocke_shared "${COMGR_LIBRARY}" "${HIP_LIBRARY}" Threads::Threads)
