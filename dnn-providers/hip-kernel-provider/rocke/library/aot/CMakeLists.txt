# Provider-owned rocKE ahead-of-time artifacts: loose per-arch HSACO + sidecars
# (leg 1), packed into per-arch .kpack + bundle manifests and installed beside
# the provider plugin data (leg 2). Producer + packaging only -- no C++.
include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/rocke_aot.cmake")

# Aggregate targets. Kernel-family registrations attach one per-arch target to
# each. rocke_client_aot_artifacts depends on rocke_client_aot_kpack so building
# artifacts also produces the .kpack + manifests.
#
# rocke_client_aot_artifacts is part of ALL: the artifacts + kpack + bundle
# manifests are a shipped product (install rules below copy the build-tree
# outputs), so they must be generated by the normal `cmake --build` before
# `cmake --install`. HSACO generation runs through comgr on the host (no GPU
# needed), so this is safe wherever rocKE is enabled.
add_custom_target(rocke_client_aot_artifacts ALL
    COMMENT "Build rocKE client AOT artifacts")
add_custom_target(rocke_client_aot_kpack
    COMMENT "Pack rocKE client AOT kpack archives + bundle manifests")
add_dependencies(rocke_client_aot_artifacts rocke_client_aot_kpack)

# Support both standalone configures (BUILD_TESTING via CTest) and
# hip-kernel-provider parent configures (HIPKERNELPROVIDER_ENABLE_TESTS).
set(_ROCKE_CLIENT_ENABLE_TESTS OFF)
if((DEFINED BUILD_TESTING AND BUILD_TESTING)
   OR (DEFINED HIPKERNELPROVIDER_ENABLE_TESTS AND HIPKERNELPROVIDER_ENABLE_TESTS))
    set(_ROCKE_CLIENT_ENABLE_TESTS ON)
endif()

# Catalog-tree path shape: kernels/<arch>/<family>/aot_list.json (matches the
# existing library/kernels/<arch>/ layout). The family handler is shared across
# arches at kernels/common/<family>_aot.py; family schemas overlay the shared
# AOT schemas at aot/schemas/<family>/.
set(_ROCKE_KERNELS_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../kernels")
set(_ROCKE_FMHA_HANDLER "${_ROCKE_KERNELS_DIR}/common/fmha_mfma_aot.py")
set(_ROCKE_FMHA_SCHEMA_DIR "${CMAKE_CURRENT_SOURCE_DIR}/schemas/fmha_fwd_mfma")

rocke_client_add_aot_instances(
    NAME sdpa_fwd_fmha_mfma_gfx942
    ARCH gfx942
    ARCH_DIR "${_ROCKE_KERNELS_DIR}/gfx942/fmha_fwd_mfma"
    HANDLER "${_ROCKE_FMHA_HANDLER}"
    SCHEMA_DIR "${_ROCKE_FMHA_SCHEMA_DIR}"
)

rocke_client_add_aot_instances(
    NAME sdpa_fwd_fmha_mfma_gfx1151
    ARCH gfx1151
    ARCH_DIR "${_ROCKE_KERNELS_DIR}/gfx1151/fmha_fwd_mfma"
    HANDLER "${_ROCKE_FMHA_HANDLER}"
    SCHEMA_DIR "${_ROCKE_FMHA_SCHEMA_DIR}"
)

if(_ROCKE_CLIENT_ENABLE_TESTS)
    rocke_client_aot_pythonpath_environment(_ROCKE_CLIENT_AOT_TEST_ENVIRONMENT)

    # Host-side Python suites: schema/sidecar/instance/handler parsers + build
    # tool + kpack round-trip/manifest. Held to the --fail-under=90 coverage
    # gate over the producer surface (see tests/coveragerc).
    add_test(
        NAME rocke_client_aot_pytest
        COMMAND "${ROCKE_PYENV_PYTHON}" -m pytest "${CMAKE_CURRENT_SOURCE_DIR}/tests"
                --cov --cov-config="${CMAKE_CURRENT_SOURCE_DIR}/tests/coveragerc"
                --cov-report=term-missing --cov-fail-under=90
        WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}"
    )
    # ROCKE_KPACK_REQUIRED makes the kpack suite fail loud (not skip) when
    # rocm_kpack is unimportable: the build environment guarantees it (configure
    # gate), so a test-time import failure means the test env plumbing regressed.
    set_tests_properties(rocke_client_aot_pytest PROPERTIES
        LABELS "rocKE-client;sdpa;unit_test"
        ENVIRONMENT "${_ROCKE_CLIENT_AOT_TEST_ENVIRONMENT};ROCKE_KPACK_REQUIRED=1"
    )

    # Numeric verifiers consume the loose build-tree artifacts and skip (CTest
    # code 77) unless the visible HIP device exactly matches --arch.
    foreach(_arch gfx942 gfx1151)
        set(_artifact_dir
            "${CMAKE_CURRENT_BINARY_DIR}/artifacts/${_arch}/sdpa_fwd_fmha_mfma_${_arch}")
        add_test(
            NAME rocke_client_sdpa_aot_numeric_${_arch}
            COMMAND "${ROCKE_PYENV_PYTHON}"
                    "${CMAKE_CURRENT_SOURCE_DIR}/tests/sdpa_aot_numeric.py"
                    --arch ${_arch}
                    --artifact-dir "${_artifact_dir}"
        )
        set_tests_properties(rocke_client_sdpa_aot_numeric_${_arch} PROPERTIES
            LABELS "gpu;rocKE-client;sdpa"
            REQUIRED_FILES "${_artifact_dir}/build.stamp"
            SKIP_RETURN_CODE 77
            ENVIRONMENT "${_ROCKE_CLIENT_AOT_TEST_ENVIRONMENT}"
        )
    endforeach()
endif()

# Convenience: build artifacts + kpack, then run only rocKE-client CTest entries.
add_custom_target(rocke_client_aot_check
    COMMAND "${CMAKE_CTEST_COMMAND}" --output-on-failure -L rocKE-client
    DEPENDS rocke_client_aot_artifacts
    COMMENT "Run rocKE client AOT tests")
