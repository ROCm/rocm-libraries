# One decoder layer of the target model as one persistent ck_tile::Megakernel launch per TP rank,
# chaining the phases of examples 58-66 (gfx125x only: scaled WMMA, async LDS loads).
# Every (small, world, KV type, kMaxM) combination is its own translation unit, generated here,
# so the 16 megakernel instantiations compile in parallel.
if(GPU_TARGETS MATCHES "gfx125")
  set(EXAMPLE_MEGAKERNEL_LAYER tile_example_megakernel_decoder_layer)
  set(_layer_srcs megakernel_decoder_layer.cpp decoder_layer_ref.cpp)
  set(_layer_gen_dir ${CMAKE_CURRENT_BINARY_DIR}/instances)
  file(MAKE_DIRECTORY ${_layer_gen_dir})
  foreach(_mm 16 64)
    foreach(_s 0 1)
      foreach(_w 1 2)
        foreach(_f 0 1)
          set(_name run_layer_s${_s}_w${_w}_f${_f}_m${_mm})
          set(_file ${_layer_gen_dir}/${_name}.cpp)
          set(_content "// Generated by example/ck_tile/67_megakernel_decoder_layer/CMakeLists.txt\n#include \"decoder_layer_device.hpp\"\nnamespace layer67 {\nint ${_name}(const LayerDims& d, const HostModel& m, const RunOptions& o, DeviceRun& r)\n{\n    return DecoderLayer<${_s} != 0, ${_w}, ${_f} != 0, ${_mm}>::run(d, m, o, r);\n}\n} // namespace layer67\n")
          if(EXISTS ${_file})
            file(READ ${_file} _old)
          else()
            set(_old "")
          endif()
          if(NOT _old STREQUAL _content)
            file(WRITE ${_file} "${_content}")
          endif()
          list(APPEND _layer_srcs ${_file})
        endforeach()
      endforeach()
    endforeach()
  endforeach()
  set_source_files_properties(${_layer_srcs} PROPERTIES LANGUAGE HIP)
  add_executable(${EXAMPLE_MEGAKERNEL_LAYER} ${_layer_srcs})
  target_include_directories(${EXAMPLE_MEGAKERNEL_LAYER} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  set(_megakernel_layer_archs ${SUPPORTED_GPU_TARGETS})
  list(FILTER _megakernel_layer_archs INCLUDE REGEX "^gfx125")
  set_property(TARGET ${EXAMPLE_MEGAKERNEL_LAYER} PROPERTY HIP_ARCHITECTURES ${_megakernel_layer_archs})
  target_compile_options(${EXAMPLE_MEGAKERNEL_LAYER} PRIVATE
      -mllvm -enable-noalias-to-md-conversion=0 "SHELL:-mllvm -enable-post-misched=1")
endif()
