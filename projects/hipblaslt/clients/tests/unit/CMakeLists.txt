# Host-only client unit tests: they exercise pure functions from
# hipblaslt_datatype2string.hpp (string<->enum parsers and the scaling-format
# helpers) and never initialize a device. "Host-only" means no device code and no
# GPU needed at run time; the TUs are still compiled as HIP, see below.
#
# These sources are compiled into two targets on purpose:
#
#   1. hipblaslt-test, so the cases run in the existing hipblaslt PR-CI lane.
#      That lane runs ctest with label selection against the *installed* tree
#      (bin/hipblaslt/CTestTestfile.cmake, generated by
#      apply_test_category_labels for hipblaslt-test only), so adding gtest cases
#      to that binary needs no install rule and no new ctest entry. See the
#      "*HostUnit*" pattern in ../test_categories.yaml, without which the cases
#      would be built but never selected.
#
#   2. hipblaslt-test-hostunit, a standalone binary that runs these cases without
#      the rest of the matmul suite, on a machine with no GPU. The top-level
#      CMakeLists.txt installs it next to hipblaslt-test so it ships in the test
#      artifact and can be pulled straight from a CI run. It is still not
#      registered with ctest: a second labelled entry would hit the multi-entry
#      sharding hazard described above.
#
#      The "hipblaslt-test" prefix is load-bearing. TheRock picks what goes in the
#      blas test artifact with a "bin/hipblaslt-test*" glob in
#      math-libs/BLAS/artifact-blas.toml, so a name outside that pattern needs a
#      matching change there. Nothing globs that pattern to decide what to *run*
#      (test_hipblaslt.py execs hipblaslt-test by exact path, and
#      apply_test_category_labels is called with that one target), so the extra
#      binary is shipped but never auto-executed.
set(HIPBLASLT_HOST_UNIT_TEST_SOURCES
    ${CMAKE_CURRENT_SOURCE_DIR}/parser_gtest.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/scaling_format_gtest.cpp
)

# hipblaslt-test is created in clients/CMakeLists.txt before add_subdirectory(tests),
# so the target already exists here.
target_sources(hipblaslt-test PRIVATE ${HIPBLASLT_HOST_UNIT_TEST_SOURCES})

# hip::device, not hip::host, even though nothing here is device code: the client
# headers require compilation as HIP. Without the `-x hip` that hip::device adds
# via INTERFACE_COMPILE_OPTIONS there is no __HIPCC__, and without __HIPCC__
# hipblaslt_float8.h hands out opaque `struct { uint8_t __x; }` stubs for the f8
# and bf8 types while auxiliary.hpp, reached through hipblaslt_datatype2string.hpp,
# calls is_nan() on them. hip::device cannot be inherited here either:
# hipblaslt-clients-common is STATIC and takes it PRIVATE, so it arrives as
# $<LINK_ONLY:hip::device>, which carries no compile options.
#
# OpenMP is omitted because it is already a PUBLIC dependency of
# hipblaslt-clients-common.
add_executable(hipblaslt-test-hostunit ${HIPBLASLT_HOST_UNIT_TEST_SOURCES})

target_link_libraries(hipblaslt-test-hostunit
    PRIVATE
        hipblaslt::hipblaslt-clients-common
        hip::device
        GTest::gtest
        GTest::gtest_main
)

target_compile_definitions(hipblaslt-test-hostunit PRIVATE GOOGLE_TEST)
